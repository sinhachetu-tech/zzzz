   254→   254→/* tiny markdown renderer — headings, bold, lists */
   255→   255→function renderMd(md: string): string {
   256→   256→  const esc = (s: string) => s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
   257→   257→  const lines = md.split("\n");
   258→   258→  let html = "";
   259→   259→  let inUl = false, inOl = false;
   260→   260→  const closeLists = () => { if (inUl) { html += "</ul>"; inUl = false; } if (inOl) { html += "</ol>"; inOl = false; } };
   261→   261→  for (const raw of lines) {
   262→   262→    const line = raw.trimEnd();
   263→   263→    if (/^##\s+/.test(line)) { closeLists(); html += `<h2>${esc(line.replace(/^##\s+/, ""))}</h2>`; continue; }
   264→   264→    if (/^#\s+/.test(line)) { closeLists(); html += `<h2>${esc(line.replace(/^#\s+/, ""))}</h2>`; continue; }
   265→   265→    if (/^\s*[-*]\s+/.test(line)) { if (!inUl) { closeLists(); html += "<ul>"; inUl = true; } html += `<li>${inline(esc(line.replace(/^\s*[-*]\s+/, "")))}</li>`; continue; }
   266→   266→    if (/^\s*\d+\.\s+/.test(line)) { if (!inOl) { closeLists(); html += "<ol>"; inOl = true; } html += `<li>${inline(esc(line.replace(/^\s*\d+\.\s+/, "")))}</li>`; continue; }
   267→   267→    if (line.trim() === "") { closeLists(); continue; }
   268→   268→    closeLists();
   269→   269→    html += `<p>${inline(esc(line))}</p>`;
   270→   270→  }
   271→   271→  closeLists();
   272→   272→  return html;
   273→   273→}
   274→   274→function inline(s: string): string {
   275→   275→  return s
   276→   276→    .replace(/`([^`]+)`/g, '<code>$1</code>')
   277→   277→    .replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>')
   278→   278→    .replace(/\*([^*]+)\*/g, '<em>$1</em>');
   279→   279→}
