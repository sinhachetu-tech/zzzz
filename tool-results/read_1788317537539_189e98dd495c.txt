    68→    68→function normalizeInput(raw: Partial<MortgageInput>): MortgageInput {
    69→    69→  const base = defaultInput();
    70→    70→  const co = raw.coBorrower as
    71→    71→    | (Partial<CoBorrower> & { existingEmis?: number })
    72→    72→    | null
    73→    73→    | undefined;
    74→    74→  const coBorrower: CoBorrower | null = co
    75→    75→    ? {
    76→    76→        name: co.name ?? "",
    77→    77→        dob: co.dob ?? "1992-01-15",
    78→    78→        incomes: Array.isArray(co.incomes) ? co.incomes : [newIncomeRow()],
    79→    79→        liabilities: Array.isArray(co.liabilities)
    80→    80→          ? co.liabilities
    81→    81→          : co.existingEmis
    82→    82→          ? [{ ...newLiabRow("Other Loan"), name: "Carried-over EMIs", method: "Manual", monthlyEmi: co.existingEmis }]
    83→    83→          : [],
    84→    84→      }
    85→    85→    : null;
    86→    86→  return {
    87→    87→    ...base,
    88→    88→    ...raw,
    89→    89→    customLtv: typeof raw.customLtv === "string" ? raw.customLtv : "",
    90→    90→    coBorrower,
    91→    91→    incomes: Array.isArray(raw.incomes) && raw.incomes.length ? raw.incomes : base.incomes,
    92→    92→    liabilities: Array.isArray(raw.liabilities) ? raw.liabilities : [],
    93→    93→  } as MortgageInput;
    94→    94→}
