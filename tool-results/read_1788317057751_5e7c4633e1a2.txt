    41→const PALETTE = ["#f2b04c", "#57c2ea", "#43d69b", "#f27363", "#b48ef2", "#8ca6b0", "#e8c15a", "#6fd3c7"];
    42→

    45→  const cases = useMemo(() => visibleCases(), [visibleCases]);
    46→  const tasks = useMemo(() => visibleTasks(), [visibleTasks]);
    47→  const statusOf = (c: LoanCase): CaseStatus => caseStatusOf(c, db.tasks);
    48→  const openTasks = tasks.filter((t) => t.status === "Open");
    49→  const today = todayISO();
    50→  const openCases = cases.filter((c) => c.caseStatus === "Active");
    51→  const booked = cases.filter((c) => c.caseStatus === "Closed");
    52→
    53→  const [bizBank, setBizBank] = useState("All");
    54→  const [bizOwner, setBizOwner] = useState("All");

    56→  /* 1 — why pending */

    70→    .filter((r) => r.s === "Overdue" || r.s === "At Risk")
    71→    .sort((a, b) => (a.nt?.dueDate ?? "9999").localeCompare(b.nt?.dueDate ?? "9999"));

    78→    })
    79→    .sort((a, b) => b.open - a.open);
    80→

    86→      const avg = inStage.length ? Math.round(inStage.reduce((sum, c) => sum + ageDays(c.createdAt), 0) / inStage.length) : 0;
    87→      return { label: s.label, value: avg, color: avg > 25 ? TONE_HEX.coral : avg > 14 ? TONE_HEX.amber : TONE_HEX.mint, sub: `${inStage.length} case${inStage.length === 1 ? "" : "s"}` };
    88→    })
    89→    .filter((r) => r.value > 0);
    90→
    91→  /* 6 — no next action */

    93→

    99→    d.setMonth(d.getMonth() - i);
   100→    months.push(d.toISOString().slice(0, 7));
