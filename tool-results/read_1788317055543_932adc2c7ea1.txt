   185→   185→function Composer({ onIssued }: { onIssued: () => void }) {
   186→   186→  const { db, session, issueBulletin, toast, visibleCases } = useStore();
   187→   187→  const me = session!;
   188→   188→  const f = flagsFor(db.designations, me.role);
   189→   189→  const teamUsers = db.users.filter((u) => u.active && u.id !== me.id && (f.scope === "all" || u.team === me.team));
   190→   190→  const [text, setText] = useState("");
   191→   191→  const [targets, setTargets] = useState<number[]>(teamUsers.map((u) => u.id));
   192→   192→  const [caseId, setCaseId] = useState<number | null>(null);
   193→   193→  const [repeat, setRepeat] = useState<"none" | "daily" | "weekdays">("none");
   194→   194→  const cases = visibleCases().filter((c) => c.caseStatus === "Active");
   195→   195→
   196→   196→  const toggle = (id: number) => setTargets((t) => (t.includes(id) ? t.filter((x) => x !== id) : [...t, id]));
   197→   197→
   198→   198→  const issue = () => {
   199→   199→    if (!text.trim()) return toast("error", "Write the directive first.");
   200→   200→    if (targets.length === 0) return toast("error", "Pick at least one teammate.");
   201→   201→    issueBulletin({ date: todayISO(), task: text, caseId, targets, repeat, asTemplate: repeat !== "none" });
   202→   202→    toast("success", repeat === "none" ? "Directive issued for today." : `Routine saved — issued today and every ${repeat === "daily" ? "day" : "working day (Mon–Fri)"}.`);
   203→   203→    setText("");
   204→   204→    setCaseId(null);
   205→   205→    setRepeat("none");
   206→   206→    onIssued();
   207→   207→  };
   208→   208→
   209→   209→  return (
   210→   210→    <div className="card p-4 anim-fade-up" style={{ borderColor: "var(--amber-line)" }}>
   211→   211→      <div className="flex items-center gap-2 mb-3">
   212→   212→        <IFlag size={15} className="text-[var(--amber)]" />

   214→   214→        <span className="text-[11.5px] text-[var(--ink-faint)] ml-auto hidden sm:inline">lands on your team's morning bulletin</span>

   217→   217→        className="textarea"
   218→   218→        rows={2}

   221→   221→        value={text}
   222→   222→        onChange={(e) => setText(e.target.value)}
   223→   223→      />
   224→   224→      <div className="flex flex-wrap items-center gap-2 mt-3">
   225→   225→        <select className="select" style={{ width: 210 }} value={caseId ?? ""} onChange={(e) => setCaseId(e.target.value ? parseInt(e.target.value, 10) : null)}>
   226→   226→          <option value="">General — no case link</option>
   227→   227→          {cases.map((c) => (
   228→   228→            <option key={c.id} value={c.id}>
   229→   229→              {c.caseNumber} · {c.customer}
   230→   230→            </option>
   231→   231→          ))}
   232→   232→        </select>
   233→   233→        <select className="select" style={{ width: 190 }} value={repeat} onChange={(e) => setRepeat(e.target.value as typeof repeat)}>
   234→   234→          <option value="none">One-off (today only)</option>
   235→   235→          <option value="daily">Repeat · every day</option>
   236→   236→          <option value="weekdays">Repeat · Mon–Fri</option>
   237→   237→        </select>
   238→   238→        <div className="flex items-center gap-1.5 ml-auto">
   239→   239→          <span className="text-[11px] uppercase tracking-[0.08em] text-[var(--ink-faint)] font-disp font-semibold">to</span>
   240→   240→          {teamUsers.map((u) => (
   241→   241→            <button
   242→   242→              key={u.id}
   243→   243→              className="chip transition-all"
   244→   244→              style={
   245→   245→                targets.includes(u.id)
   246→   246→                  ? { background: "var(--amber-tint)", borderColor: "var(--amber)", color: "var(--amber)" }
   247→   247→                  : { background: "var(--bg2)", borderColor: "var(--line)", color: "var(--ink-faint)" }

   249→   249→              onClick={() => toggle(u.id)}

   257→   257→        <p className="text-[11px] text-[var(--ink-faint)] m-0">

   261→   261→          <IPlus size={13} /> Issue{repeat !== "none" ? " routine" : ""}

   270→   270→function Scoreboard({ items }: { items: BulletinItem[] }) {
   271→   271→  const { db, userById } = useStore();
   272→   272→  const rows = useMemo(() => {
   273→   273→    const map = new Map<number, { done: number; total: number }>();
   274→   274→    for (const b of items) {
   275→   275→      if (b.isTemplate || b.dropped) continue;
   276→   276→      for (const t of b.targets) {
   277→   277→        const cur = map.get(t) ?? { done: 0, total: 0 };
   278→   278→        cur.total += 1;
   279→   279→        if (b.status === "Done") cur.done += 1;
   280→   280→        map.set(t, cur);
   281→   281→      }
   282→   282→    }
   283→   283→    return [...map.entries()]
   284→   284→      .map(([id, v]) => ({ id, name: userById(id)?.name ?? "?", ...v }))
   285→   285→      .sort((a, b) => b.total - a.total);
   286→   286→  }, [items, userById]);
   287→   287→
   288→   288→  if (rows.length === 0) return null;
   289→   289→  return (
   290→   290→    <div className="card p-4 anim-fade-up">
   291→   291→      <h3 className="font-disp font-semibold text-[13.5px] m-0 mb-1">Team pulse · today</h3>
   292→   292→      <p className="text-[11px] text-[var(--ink-faint)] m-0 mb-3">directives cleared per teammate</p>
   293→   293→      <div className="space-y-2.5">
   294→   294→        {rows.map((r) => {
   295→   295→          const pct = r.total ? (r.done / r.total) * 100 : 0;
   296→   296→          const clean = r.done === r.total && r.total > 0;
   297→   297→          return (
   298→   298→            <div key={r.id} className="flex items-center gap-2.5">
   299→   299→              <Avatar name={r.name} size={24} />
   300→   300→              <div className="flex-1 min-w-0">
   301→   301→                <div className="flex justify-between items-baseline mb-0.5">
   302→   302→                  <span className="text-[12px] truncate">{r.name}</span>
   303→   303→                  <span className="mono text-[11px]" style={{ color: clean ? "var(--mint)" : pct < 50 ? "var(--coral)" : "var(--ink-dim)" }}>
   304→   304→                    {r.done}/{r.total}
   305→   305→                  </span>
   306→   306→                </div>
   307→   307→                <div className="h-[5px] rounded-full overflow-hidden" style={{ background: "var(--track)" }}>
   308→   308→                  <div className="h-full rounded-full" style={{ width: `${pct}%`, background: clean ? "var(--mint)" : pct < 50 ? "var(--coral)" : "var(--amber)", transition: "width 0.7s cubic-bezier(0.22,1,0.36,1)" }} />
   309→   309→                </div>
   310→   310→              </div>
   311→   311→            </div>
   312→   312→          );
   313→   313→        })}
   314→   314→      </div>
   315→   315→    </div>
   316→   316→  );
   317→   317→}

   344→   344→    .filter((b) => b.date < today || b.dropped || (b.date === today && b.status === "Done"))
   345→   345→    .sort((a, b) => b.date.localeCompare(a.date) || b.createdAt.localeCompare(a.createdAt));

   347→   347→    () => (me ? db.bulletin.filter((b) => b.isTemplate && (bulletinCanManage(b, me, db) || bulletinVisible(b, me, db))) : []),

   350→   350→
   351→   351→  if (!me) return null;
   352→   352→
   353→   353→  const archiveGroups = new Map<string, BulletinItem[]>();
   354→   354→  for (const b of archiveItems) {
   355→   355→    const list = archiveGroups.get(b.date) ?? [];
   356→   356→    list.push(b);
   357→   357→    archiveGroups.set(b.date, list);

   363→   363→      "directive-archive.csv",
   364→   364→      ["Date", "Task", "Issued by", "Role", "Targets", "Case", "Status", "Completed"],

   370→   370→        b.targets.map((t) => userById(t)?.name ?? "").join("; "),
   371→   371→        b.caseId ? db.cases.find((c) => c.id === b.caseId)?.caseNumber ?? "" : "",
   372→   372→        b.dropped ? "Dropped" : b.status,
   373→   373→        b.completedAt ? relTime(b.completedAt) : "",
   374→   374→      ])
