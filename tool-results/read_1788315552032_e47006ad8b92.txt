    52→const SUPER_FLAGS: RoleFlags = { scope: "all", issueTasks: true, admin: true, super: true };
    53→const DEFAULT_FLAGS: RoleFlags = { scope: "own", issueTasks: false, admin: false, super: false };
    54→
    55→export function flagsFor(designations: Designation[], role: string): RoleFlags {
    56→  if (role === "Super Admin") return SUPER_FLAGS;
    57→  const d = designations.find((x) => x.name === role);
    58→  return d ? { scope: d.scope, issueTasks: d.issueTasks, admin: d.admin, super: d.super } : DEFAULT_FLAGS;
    59→}

    61→/* who can see / act on a bulletin directive */
    62→export function bulletinVisible(b: BulletinItem, me: User, db: DB): boolean {
    63→  const f = flagsFor(db.designations, me.role);
    64→  if (f.scope === "all") return true;
    65→  if (b.issuedBy === me.id || b.targets.includes(me.id)) return true;
    66→  if (f.scope === "team") {
    67→    const issuer = db.users.find((u) => u.id === b.issuedBy);
    68→    if (issuer?.team === me.team) return true;
    69→    return b.targets.some((t) => db.users.find((u) => u.id === t)?.team === me.team);
    70→  }
    71→  return false;
    72→}

    74→export function bulletinCanAct(b: BulletinItem, me: User, db: DB): boolean {
    75→  const f = flagsFor(db.designations, me.role);
    76→  return f.scope === "all" || b.issuedBy === me.id || b.targets.includes(me.id);

    91→const MS_DAY = 86400000;
    92→const SPAWN_BACK_LIMIT = 3; // don't drown a returning user: at most 3 past instances per template
    93→
    94→function eligibleDays(fromISO: string, repeat: "daily" | "weekdays"): string[] {
    95→  const out: string[] = [];
    96→  const today = todayISO();
    97→  const span = Math.min(daysBetween(fromISO, today), SPAWN_BACK_LIMIT + 1);
    98→  for (let back = Math.max(0, span); back >= 0; back--) {
    99→    const d = new Date(Date.now() - back * MS_DAY);
   100→    const iso = toISODate(d);
   101→    if (iso < fromISO) continue;
   102→    if (repeat === "weekdays") {
   103→      const dow = d.getDay();
   104→      if (dow === 0 || dow === 6) continue; // UAE weekend: Sat–Sun
   105→    }
   106→    out.push(iso);
   107→  }
   108→  return out;
   109→}

   111→export function spawnBulletinInstances(prev: DB): DB {
   112→  const templates = prev.bulletin.filter((b) => b.isTemplate && b.repeat && b.repeat !== "none");
   113→  if (templates.length === 0) return prev;
   114→  let nid = prev.bulletin.reduce((m, b) => Math.max(m, b.id), 0) + 1;
   115→  const added: BulletinItem[] = [];
   116→  for (const t of templates) {
   117→    for (const day of eligibleDays(t.date, t.repeat as "daily" | "weekdays")) {
   118→      const exists = prev.bulletin.some((b) => b.templateId === t.id && b.date === day) || added.some((b) => b.templateId === t.id && b.date === day);
   119→      if (exists) continue;
   120→      added.push({
   121→        ...t,
   122→        id: nid++,
   123→        date: day,
   124→        templateId: t.id,
   125→        isTemplate: false,
   126→        repeat: "none",
   127→        status: "Open",
   128→        completedAt: null,
   129→        completedBy: null,
   130→        carriedFrom: null,
   131→        dropped: false,
   132→        createdAt: nowISO(),
   133→        replies: [],
   134→      });
   135→    }
   136→  }
   137→  return added.length ? { ...prev, bulletin: [...prev.bulletin, ...added] } : prev;
   138→}

   140→export function resolveStaleBulletins(prev: DB): DB {
   141→  const casesById = new Map(prev.cases.map((c) => [c.id, c]));
   142→  let rid = prev.bulletin.reduce((m, b) => Math.max(m, b.id, ...b.replies.map((r) => r.id)), 0) + 1;
   143→  let changed = false;
   144→  const bulletin = prev.bulletin.map((b) => {
   145→    if (b.caseId != null && b.status === "Open" && !b.isTemplate) {
   146→      const c = casesById.get(b.caseId);
   147→      if (!c || c.caseStatus !== "Active") {
   148→        changed = true;
   149→        return {
   150→          ...b,
   151→          status: "Done" as const,
   152→          completedAt: nowISO(),
   153→          completedBy: null,
   154→          replies: [...b.replies, { id: rid++, userId: b.issuedBy, text: "Case closed — directive resolved automatically, no action needed.", at: nowISO() }],
   155→        };
   156→      }
   157→    }
   158→    return b;
   159→  });
   160→  return changed ? { ...prev, bulletin } : prev;
   161→}

   224→function parseHash(): Route {
   225→  const h = window.location.hash.replace(/^#\/?/, "");
   226→  const [a, b] = h.split("/");
   227→  if (a === "case" && b && !Number.isNaN(parseInt(b, 10))) return { name: "case", id: parseInt(b, 10) };
   228→  if (a === "tasks") return { name: "tasks" };
   229→  if (a === "bulletin") return { name: "bulletin" };
   230→  if (a === "calculator") return { name: "calculator" };
   231→  if (a === "reports") return { name: "reports" };
   232→  if (a === "admin") return { name: "admin" };
   233→  return { name: "dashboard" };
   234→}

   236→export function routeToHash(r: Route): string {
   237→  switch (r.name) {
   238→    case "case":
   239→      return `#/case/${r.id}`;
   240→    case "dashboard":
   241→      return "#/";
   242→    default:
   243→      return `#/${r.name}`;
   244→  }
   245→}

   250→function loadDb(): DB {
   251→  try {
   252→    const raw = localStorage.getItem(DB_KEY);
   253→    if (raw) {
   254→      const parsed = JSON.parse(raw) as DB;
   255→      if (parsed && parsed.version === 9 && Array.isArray(parsed.designations) && Array.isArray(parsed.bulletin)) return parsed;
   256→    }
   257→  } catch {
   258→    /* fall through to seed */
   259→  }
   260→  return seedDb();
   261→}

   263→export function StoreProvider({ children }: { children: ReactNode }) {
   264→  const [db, setDb] = useState<DB>(loadDb);
   265→  const [sessionId, setSessionId] = useState<number | null>(() => {
   266→    const raw = localStorage.getItem(SESSION_KEY);

   270→  const [toasts, setToasts] = useState<ToastMsg[]>([]);
   271→  const toastId = useRef(0);
   272→
   273→  /* Boot: apply the bulletin lifecycle (spawn routines / resolve stale) to whatever
   274→     state we start with. When Supabase is configured, hydrate from the shared database
   275→     first, then apply the same lifecycle — so the whole team shares one source of truth. */
   276→  useEffect(() => {
   277→    let cancelled = false;
   278→    if (isSupabaseOn) {
   279→      fetchAppState().then((server) => {
   280→        if (cancelled) return;
   281→        setDb((prev) => resolveStaleBulletins(spawnBulletinInstances(server ?? prev)));
   282→      });
   283→    } else {
   284→      setDb((prev) => resolveStaleBulletins(spawnBulletinInstances(prev)));
   285→    }
   286→    return () => {
   287→      cancelled = true;
   288→    };

   290→  }, []);
   291→
   292→  /* Persist on every change: always keep a local copy (so the no-Supabase build works
   293→     and survives refresh), and when Supabase is configured, write the shared document
   294→     too — debounced so rapid edits batch into one save. */
   295→  useEffect(() => {
   296→    try {
   297→      localStorage.setItem(DB_KEY, JSON.stringify(db));
   298→    } catch {
   299→      /* storage full / private mode — non-fatal */
   300→    }

   307→
   308→  useEffect(() => {
   309→    const onHash = () => setRoute(parseHash());

   311→    return () => window.removeEventListener("hashchange", onHash);
   312→  }, []);
   313→
   314→  const nav = useCallback((r: Route) => {
   315→    const h = routeToHash(r);
   316→    if (window.location.hash === h) setRoute(r);
   317→    else window.location.hash = h;
   318→  }, []);
   319→
   320→  const toast = useCallback((kind: ToastMsg["kind"], msg: string) => {
   321→    toastId.current += 1;
   322→    const id = toastId.current;
   323→    setToasts((t) => [...t, { id, kind, msg }]);

   326→
   327→  const dismissToast = useCallback((id: number) => {

   329→  }, []);
   330→
   331→  const session = sessionId != null ? db.users.find((u) => u.id === sessionId) ?? null : null;
   332→  const userById = useCallback((id: number) => db.users.find((u) => u.id === id), [db]);

   334→  const logAct = (list: Activity[], caseId: number, userId: number, action: string, oldValue?: string, newValue?: string): Activity[] => [

   481→  const updateCase = useCallback(
   482→    (id: number, patch: Partial<Omit<LoanCase, "id" | "createdAt">>) => {
   483→      const me = session;
   484→      setDb((prev) => {
   485→        const before = prev.cases.find((c) => c.id === id);
   486→        if (!before) return prev;
   487→        let acts = prev.activities;
   488→        if (patch.stage !== undefined && patch.stage !== before.stage)
   489→          acts = logAct(acts, id, me?.id ?? 0, "Stage moved", before.stage, patch.stage);
   490→        if (patch.ownerId !== undefined && patch.ownerId !== before.ownerId)
   491→          acts = logAct(acts, id, me?.id ?? 0, "Owner changed",
   492→            prev.users.find((u) => u.id === before.ownerId)?.name ?? "—",

  1290→export function computeKpis(
  1291→  cases: LoanCase[],
  1292→  tasks: Task[],
  1293→  statusOf: (c: LoanCase) => "On Track" | "At Risk" | "Overdue" | "No Action",
  1294→  banks: BankItem[],
  1295→  escalations: number
  1296→): Kpis {
  1297→  const open = cases.filter((c) => c.caseStatus === "Active");
  1298→  let overdue = 0, atRisk = 0, noAction = 0, estCommission = 0;
  1299→  for (const c of open) {
  1300→    const s = statusOf(c);
  1301→    if (s === "Overdue") overdue += 1;
  1302→    else if (s === "At Risk") atRisk += 1;
  1303→    else if (s === "No Action") noAction += 1;
  1304→    const b = banks.find((x) => x.name === primaryBank(c));
  1305→    estCommission += b ? (c.loanAmount * b.ratePct) / 100 : 0;
  1306→  }
  1307→  const openTasks = tasks.filter((t) => t.status === "Open");
  1308→  return {
  1309→    openCases: open.length,
  1310→    overdue,
  1311→    atRisk,
  1312→    noAction,
  1313→    openTasks: openTasks.length,
  1314→    dueToday: openTasks.filter((t) => t.dueDate === todayISO()).length,
  1315→    pipelineValue: open.reduce((s, c) => s + c.loanAmount, 0),
  1316→    estCommission: Math.round(estCommission),
  1317→    escalations,
  1318→  };
  1319→}

  1321→export function activityPerDay(activities: Activity[], days: number): number[] {
  1322→  const out = new Array(days).fill(0) as number[];
  1323→  const today = new Date();
  1324→  for (const a of activities) {
  1325→    const d = new Date(a.at);
  1326→    const diff = Math.floor((today.setHours(0, 0, 0, 0) - d.setHours(0, 0, 0, 0)) / 86400000);
  1327→    if (diff >= 0 && diff < days) out[days - 1 - diff] += 1;
  1328→  }
  1329→  return out;
  1330→}

  1332→export function stageEnteredAt(c: LoanCase, activities: Activity[]): string {
  1333→  const moves = activities
  1334→    .filter((a) => a.caseId === c.id && a.action === "Stage moved" && a.newValue === c.stage)
  1335→    .sort((a, b) => b.at.localeCompare(a.at));
  1336→  return moves[0]?.at ?? c.createdAt;
  1337→}

  1339→export function slaFor(rules: SlaRule[], stage: string, bank: string | null): SlaRule | null {
  1340→  const candidates = rules.filter((r) => r.active && r.stage === stage);
  1341→  return candidates.find((r) => r.bank === bank) ?? candidates.find((r) => r.bank === null) ?? null;
  1342→}

  1344→export interface Escalation {
  1345→  c: LoanCase;
  1346→  rule: SlaRule;
  1347→  days: number;
  1348→}

  1350→export function computeEscalations(db: DB, cases: LoanCase[]): Escalation[] {
  1351→  const out: Escalation[] = [];
  1352→  for (const c of cases) {
  1353→    if (c.caseStatus !== "Active") continue;
  1354→    const bank = primaryBank(c);
  1355→    const rule = slaFor(db.slaRules, c.stage, bank);
  1356→    if (!rule) continue;
  1357→    const days = Math.max(0, daysBetween(stageEnteredAt(c, db.activities).slice(0, 10), todayISO()));
  1358→    if (days > rule.maxDays) out.push({ c, rule, days });
  1359→  }
  1360→  return out.sort((a, b) => b.days - a.days);
  1361→}
