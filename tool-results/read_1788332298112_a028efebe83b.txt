    93→export default function Reports() {
    94→  const {
    95→    cases, tasks, activities, banks, partners, users, stages, slaRules,
    96→    nav, userById, visibleCases, visibleTasks, toast,
    97→  } = useHfmcStore();
    98→
    99→  const visCases = useMemo(() => visibleCases(), [visibleCases]);
   100→  const visTasks = useMemo(() => visibleTasks(), [visibleTasks]);
   101→
   102→  /* ---- slice cases by lifecycle state ---- */
   103→  const active = useMemo(() => visCases.filter((c) => c.caseStatus === "Active"), [visCases]);
   104→  const booked = useMemo(() => visCases.filter((c) => c.caseStatus === "Closed"), [visCases]);
   105→  const lost = useMemo(() => visCases.filter((c) => c.caseStatus === "Lost"), [visCases]);
   106→
   107→  /* ---- 1 · pipeline by stage ---- */
   108→  const sortedStages = useMemo(
   109→    () => [...stages].sort((a, b) => a.sortOrder - b.sortOrder),
   110→    [stages],
   111→  );
   112→  const stageRows = useMemo(
   113→    () =>
   114→      sortedStages
   115→        .map((s, i) => {
   116→          const inStage = active.filter((c) => c.stage === s.label);
   117→          const value = inStage.reduce((sum, c) => sum + c.loanAmount, 0);
   118→          return {
   119→            label: s.label,
   120→            value: inStage.length,
   121→            color: STAGE_PALETTE[i % STAGE_PALETTE.length],
   122→            sub: inStage.length ? `· ${fmtMoney(value)}` : "",
   123→            amount: value,
   124→          };
   125→        })
   126→        .filter((r) => r.value > 0),
   127→    [sortedStages, active],
   128→  );
   129→
   130→  /* ---- 2 · source mix ---- */
   131→  const sourceSegs = useMemo(
   132→    () =>
   133→      SOURCES
   134→        .map((src) => ({
   135→          label: src,
   136→          value: visCases.filter((c) => c.source === src).length,
   137→          color: SOURCE_COLORS[src],
   138→        }))
   139→        .filter((s) => s.value > 0),
   140→    [visCases],
   141→  );
   142→
   143→  /* ---- 3 · bank win rate ---- */
   144→  const bankRows = useMemo(
   145→    () =>
   146→      banks
   147→        .map((b) => {
   148→          const submitted = visCases.filter((c) => c.banks.includes(b.name));
   149→          const won = booked.filter((c) => c.wonBank === b.name);
   150→          const lostCount = lost.filter((c) => c.banks.includes(b.name)).length;
   151→          const decided = won.length + lostCount;
   152→          const winPct = decided > 0 ? Math.round((won.length / decided) * 100) : 0;
   153→          const grossEarned = won.reduce((s, c) => s + commissionFor(c, banks).gross, 0);
   154→          return { b, submitted: submitted.length, won: won.length, lostCount, winPct, grossEarned };
   155→        })
   156→        .filter((r) => r.submitted > 0 || r.won > 0)
   157→        .sort((a, b) => b.grossEarned - a.grossEarned),
   158→    [banks, visCases, booked, lost],
   159→  );
   160→
   161→  /* ---- 4 · owner leaderboard ---- */
   162→  const owners = useMemo(
   163→    () =>
   164→      users
   165→        .filter((u) => u.role !== "Head of Company" && u.role !== "PA to HoC")
   166→        .map((u) => {
   167→          const mine = visCases.filter((c) => c.ownerId === u.id);
   168→          const won = mine.filter((c) => c.caseStatus === "Closed");
   169→          const lostMine = mine.filter((c) => c.caseStatus === "Lost").length;
   170→          const decided = won.length + lostMine;
   171→          const winPct = decided > 0 ? Math.round((won.length / decided) * 100) : 0;
   172→          const bookedValue = won.reduce((s, c) => s + c.loanAmount, 0);
   173→          const netEarn = won.reduce((s, c) => s + commissionFor(c, banks).net, 0);
   174→          return { u, mine: mine.length, won: won.length, lostMine, winPct, bookedValue, netEarn };
   175→        })
   176→        .filter((r) => r.mine > 0)
   177→        .sort((a, b) => b.bookedValue - a.bookedValue),
   178→    [users, visCases, banks],
   179→  );
   180→
   181→  /* ---- 5 · conversion funnel ---- */
   182→  const funnel = useMemo(() => {
   183→    const total = visCases.length;
   184→    const bookedN = booked.length;
   185→    const lostN = lost.length;
   186→    const activeN = active.length;
   187→    const decided = bookedN + lostN;
   188→    const conv = total > 0 ? Math.round((bookedN / total) * 100) : 0;
   189→    const hitRate = decided > 0 ? Math.round((bookedN / decided) * 100) : 0;
   190→    return { total, bookedN, lostN, activeN, conv, hitRate };
   191→  }, [visCases, booked, lost, active]);
   192→
   193→  /* ---- 6 · commission summary ---- */
   194→  const commission = useMemo(() => {
   195→    const gross = booked.reduce((s, c) => s + commissionFor(c, banks).gross, 0);
   196→    const partnerCut = booked.reduce((s, c) => s + commissionFor(c, banks).partnerCut, 0);
   197→    const net = gross - partnerCut;
   198→    const pipelineGross = active.reduce((s, c) => s + commissionFor(c, banks).gross, 0);
   199→    return { gross, partnerCut, net, pipelineGross };
   200→  }, [booked, active, banks]);
   201→
   202→  /* ---- 7 · SLA breaches ---- */
   203→  const escalations = useMemo(
   204→    () => computeEscalations(visCases, slaRules),
   205→    [visCases, slaRules],
   206→  );

   208→    () =>
   209→      escalations
   210→        .map((e) => ({ e, c: visCases.find((c) => c.id === e.caseId) }))
   211→        .filter((r): r is { e: typeof escalations[number]; c: LoanCase } => !!r.c)
   212→        .sort((a, b) => b.e.breachDays - a.e.breachDays),
   213→    [escalations, visCases],
   214→  );
   215→
   216→  /* ---- 8 · activity trend (14 days) ---- */
   217→  const trend = useMemo(() => activityPerDay(activities, 14), [activities]);
   218→  const trendTotal = useMemo(() => trend.reduce((s, n) => s + n, 0), [trend]);
   219→
   220→  /* ---- CSV exports ---- */
   221→  const exportCases = () => {
   222→    const header = ["Case #", "Customer", "Status", "Stage", "Source", "Banks", "Won bank", "Loan amount (AED)", "Owner", "Partner", "Share %", "Created", "Closed"];
   223→    const rows = visCases.map((c) => [
   224→      c.caseNumber, c.customer, c.caseStatus, c.stage, c.source,
   225→      c.banks.join(" / ") || "TBC", c.wonBank ?? "", Math.round(c.loanAmount),
   226→      userById(c.ownerId)?.name ?? "", c.partner?.name ?? "", c.partner?.sharePct ?? "",
   227→      c.createdAt.slice(0, 10), c.closedDate ?? "",
   228→    ]);
   229→    downloadCSV("hfmc-cases.csv", header, rows);
   230→    toast("success", "Cases exported.");
   231→  };
   232→
   233→  const exportTasks = () => {
   234→    const header = ["Case #", "Task", "Owner", "Status", "Waiting for", "Why pending", "Due", "Created"];
   235→    const rows = visTasks.map((t) => {
   236→      const c = cases.find((x) => x.id === t.caseId);
   237→      return [
   238→        c?.caseNumber ?? "", t.description, userById(t.ownerId)?.name ?? "",
   239→        t.status, t.waitingFor, t.whyPending, t.dueDate, t.createdAt.slice(0, 10),
   240→      ];
   241→    });
   242→    downloadCSV("hfmc-tasks.csv", header, rows);
   243→    toast("success", "Tasks exported.");
   244→  };
   245→
   246→  const exportCommission = () => {
   247→    const header = ["Case #", "Customer", "Won bank", "Rate %", "Loan amount (AED)", "Gross (AED)", "Partner", "Share %", "Partner cut (AED)", "Net (AED)", "Closed"];
   248→    const rows = booked.map((c) => {
   249→      const m = commissionFor(c, banks);
   250→      return [
   251→        c.caseNumber, c.customer, m.bank ?? "", m.ratePct, Math.round(c.loanAmount),
   252→        Math.round(m.gross), c.partner?.name ?? "", c.partner?.sharePct ?? "",
   253→        Math.round(m.partnerCut), Math.round(m.net), c.closedDate ?? "",
   254→      ];
   255→    });
   256→    downloadCSV("hfmc-commission.csv", header, rows);
   257→    toast("success", "Commission exported.");
   258→  };
   259→
   260→  /* ---- render ---- */
   261→
   262→  return (
   263→    <div className="space-y-5">
   264→      {/* header */}
   265→      <div className="flex flex-wrap items-end justify-between gap-3 anim-fade-up">
   266→        <div>
   267→          <h1 className="font-disp font-bold text-[24px] tracking-tight m-0 flex items-center gap-2.5">
   268→            <IChart size={22} className="text-[var(--amber)]" /> Reports
   269→          </h1>
   270→          <p className="text-[13px] text-[var(--ink-dim)] mt-0.5 mb-0">
   271→            {visCases.length} cases · {booked.length} booked · {fmtMoney(booked.reduce((s, c) => s + c.loanAmount, 0))} closed volume
   272→          </p>
   273→        </div>
   274→        <div className="flex flex-wrap items-center gap-2">
   275→          <button className="btn btn-ghost btn-sm" onClick={exportCases} title="Export every visible case">
   276→            <IDownload size={14} /> Cases
   277→          </button>
   278→          <button className="btn btn-ghost btn-sm" onClick={exportTasks} title="Export every visible task">
   279→            <IDownload size={14} /> Tasks
   280→          </button>
   281→          <button className="btn btn-primary btn-sm" onClick={exportCommission} title="Export commission breakdown for booked cases">
   282→            <IDownload size={14} /> Commission
   283→          </button>
   284→        </div>
   285→      </div>
   286→
   287→      {/* KPI strip */}
   288→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 stagger">
   289→        <KpiCard label="Active pipeline" value={<CountUp target={active.length} />} sub={fmtMoney(active.reduce((s, c) => s + c.loanAmount, 0))} tone="amber" icon={<IBriefcase size={15} />} />
   290→        <KpiCard label="Booked" value={<CountUp target={booked.length} />} sub={fmtMoney(booked.reduce((s, c) => s + c.loanAmount, 0))} tone="mint" icon={<ITrophy size={15} />} />
   291→        <KpiCard label="Lost" value={<CountUp target={lost.length} />} sub={`${funnel.hitRate}% hit rate`} tone="coral" icon={<ITarget size={15} />} />
   292→        <KpiCard label="Net commission" value={<CountUp target={Math.round(commission.net)} format={fmtMoney} />} sub={`of ${fmtMoney(commission.gross)} gross`} tone="sky" icon={<IBank size={15} />} />
   293→      </div>
   294→
   295→      {/* report grid */}
   296→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   297→        <GroupLabel>Pipeline</GroupLabel>
   298→
   299→        {/* 1 · pipeline by stage */}
   300→        <ReportCard
   301→          title="Pipeline by stage"
   302→          sub="Live cases and loan amount per stage"
   303→          icon={<IBriefcase size={15} />}
   304→        >
   305→          {stageRows.length ? (
   306→            <BarList items={stageRows} />
   307→          ) : (
   308→            <p className="text-[12.5px] text-[var(--ink-faint)] m-0">No active cases in any stage.</p>
   309→          )}
   310→        </ReportCard>
   311→
   312→        {/* 2 · source mix */}
   313→        <ReportCard
   314→          title="Source mix"
   315→          sub="Where the case introductions come from"
   316→          icon={<IUsers size={15} />}
   317→        >
   318→          {sourceSegs.length ? (
   319→            <Donut segments={sourceSegs} centerLabel="cases" />

   321→            <p className="text-[12.5px] text-[var(--ink-faint)] m-0">No cases to break down.</p>

   323→        </ReportCard>
   324→
   325→        {/* 3 · bank win rate */}
   326→        <ReportCard
   327→          title="Bank win rate"
   328→          sub="Submitted vs booked per bank, and commission earned"
   329→          icon={<IBank size={15} />}

   335→            <div className="overflow-x-auto">

   350→                      <td className="font-medium">{r.b.name}</td>
   351→                      <td className="mono">{fmtRate(r.b.ratePct)}</td>

   367→
   368→        <GroupLabel>People</GroupLabel>
   369→
   370→        {/* 4 · owner leaderboard */}
   371→        <ReportCard
   372→          title="Owner leaderboard"
   373→          sub="Ranked by booked volume — with win rate and net commission"
   374→          icon={<ITrophy size={15} />}
   375→          span
   376→        >
   377→          {owners.length === 0 ? (
   378→            <p className="text-[12.5px] text-[var(--ink-faint)] m-0">No cases assigned yet.</p>

   393→                <tbody>
