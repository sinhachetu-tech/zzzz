   346→function MasterTab({ kind }: { kind: "stages" | "whyPending" | "waitingFor" }) {
   347→  const { db, addMaster, toggleMaster, deleteMaster, moveStage, toast } = useStore();
   348→  const [label, setLabel] = useState("");
   349→  const title = kind === "stages" ? "Workflow stages" : kind === "whyPending" ? "Why pending reasons" : "Waiting-for types";
   350→  const items = kind === "stages" ? [...db.stages].sort((a, b) => a.sortOrder - b.sortOrder) : db[kind];
   351→
   352→  return (
   353→    <div className="card anim-fade-up">
   354→      <div className="px-4 py-3 border-b" style={{ borderColor: "var(--line-soft)" }}>
   355→        <h3 className="font-disp font-semibold text-[14px] m-0">{title}</h3>
   356→        <p className="text-[11.5px] text-[var(--ink-faint)] mt-0.5 mb-0">
   357→          {kind === "stages" ? "Ordered left-to-right on every Case 360. Deactivated stages disappear from pickers." : "Used across the task engine and reports. Deactivated items stop appearing in dropdowns."}
   358→        </p>
   359→      </div>
   360→      <div className="divide-y" style={{ borderColor: "var(--line-soft)" }}>
   361→        {items.map((m) => (
   362→          <div key={m.id} className="px-4 py-2.5 flex items-center gap-3" style={{ borderColor: "var(--line-soft)", opacity: m.active ? 1 : 0.5 }}>
   363→            {kind === "stages" && (
   364→              <>
   365→                <span className="mono text-[11px] text-[var(--ink-faint)] w-6">{kind === "stages" ? (m as unknown as { sortOrder: number }).sortOrder : ""}</span>
   366→                <button className="btn btn-ghost btn-sm !px-1.5 !py-1" onClick={() => moveStage(m.id, -1)}><IChevronL size={13} /></button>
   367→                <button className="btn btn-ghost btn-sm !px-1.5 !py-1" onClick={() => moveStage(m.id, 1)}><IChevronR size={13} /></button>
   368→              </>
   369→            )}
   370→            <span className="text-[13px] flex-1">{m.label}</span>
   371→            <button className="btn btn-ghost btn-sm !px-2" title={m.active ? "Deactivate" : "Activate"} onClick={() => toggleMaster(kind, m.id)}>
   372→              <span className="w-2 h-2 rounded-full" style={{ background: m.active ? "var(--mint)" : "var(--ink-faint)" }} />
   373→            </button>
   374→            <button className="btn btn-ghost btn-sm !px-2 text-[var(--ink-faint)] hover:text-[var(--coral)]" onClick={() => {
   375→              const err = deleteMaster(kind, m.id);
   376→              if (err) toast("error", err);
   377→              else toast("success", `"${m.label}" deleted.`);
   378→            }}>
   379→              <IX size={13} />
   380→            </button>
   381→          </div>
   382→        ))}
   383→      </div>
   384→      <div className="flex items-end gap-2 px-4 py-3 border-t" style={{ borderColor: "var(--line-soft)" }}>
   385→        <div className="flex-1">
   386→          <label className="label">New label</label>
   387→          <input className="input" placeholder="e.g. Awaiting NOC from developer" value={label} onChange={(e) => setLabel(e.target.value)} onKeyDown={(e) => {
   388→            if (e.key === "Enter") {
   389→              const err = addMaster(kind, label);
   390→              if (err) toast("error", err);
   391→              else { toast("success", `"${label.trim()}" added.`); setLabel(""); }
   392→            }
   393→          }} />
   394→        </div>
   395→        <button className="btn btn-primary btn-sm" onClick={() => {
   396→          const err = addMaster(kind, label);
   397→          if (err) toast("error", err);
   398→          else { toast("success", `"${label.trim()}" added.`); setLabel(""); }
   399→        }}>
   400→          <IPlus size={13} /> Add
   401→        </button>
   402→      </div>
   403→    </div>
   404→  );
   405→}

   409→function SlaTab() {
   410→  const { db, saveSla, toggleSla, deleteSla, toast } = useStore();
   411→  const stages = [...db.stages].filter((s) => s.label !== "Closure" && s.label !== "Closed").sort((a, b) => a.sortOrder - b.sortOrder);
   412→  const [stage, setStage] = useState(stages[0]?.label ?? "");
   413→  const [bank, setBank] = useState("All");
   414→  const [days, setDays] = useState("5");
   415→
   416→  return (
   417→    <div className="card anim-fade-up">
   418→      <div className="px-4 py-3 border-b" style={{ borderColor: "var(--line-soft)" }}>
   419→        <h3 className="font-disp font-semibold text-[14px] m-0">SLA rules</h3>
   420→        <p className="text-[11.5px] text-[var(--ink-faint)] mt-0.5 mb-0">Max days a case may sit in a stage before it escalates. Bank-specific rules override the default.</p>
   421→      </div>
   422→      <div className="divide-y" style={{ borderColor: "var(--line-soft)" }}>
   423→        {[...db.slaRules].sort((a, b) => a.stage.localeCompare(b.stage)).map((r) => (
   424→          <div key={r.id} className="px-4 py-2.5 flex items-center gap-3" style={{ borderColor: "var(--line-soft)", opacity: r.active ? 1 : 0.5 }}>
   425→            <Chip tone="slate">{r.stage}</Chip>
   426→            {r.bank ? <Chip tone="sky">{r.bank}</Chip> : <span className="text-[11.5px] text-[var(--ink-faint)]">all banks</span>}
   427→            <span className="ml-auto text-[12.5px] text-[var(--ink-dim)]">max</span>
   428→            <input
   429→              className="input mono !w-[64px] !py-1 text-[12px] text-center"
   430→              type="number"
   431→              min={1}
   432→              value={r.maxDays}
   433→              onChange={(e) => saveSla({ ...r, maxDays: Math.max(1, Number(e.target.value) || 1) })}
   434→            />
   435→            <span className="text-[12px] text-[var(--ink-faint)]">days</span>
   436→            <button className="btn btn-ghost btn-sm !px-2" title={r.active ? "Deactivate" : "Activate"} onClick={() => toggleSla(r.id)}>
   437→              <span className="w-2 h-2 rounded-full" style={{ background: r.active ? "var(--mint)" : "var(--ink-faint)" }} />
   438→            </button>
   439→            <button className="btn btn-ghost btn-sm !px-2 text-[var(--ink-faint)] hover:text-[var(--coral)]" onClick={() => { deleteSla(r.id); toast("info", "SLA rule removed."); }}>
   440→              <IX size={13} />
   441→            </button>
   442→          </div>
   443→        ))}
   444→        {db.slaRules.length === 0 && <p className="text-[12.5px] text-[var(--ink-faint)] p-4 m-0">No SLA rules — nothing will escalate.</p>}
   445→      </div>
   446→      <div className="flex flex-wrap items-end gap-2 px-4 py-3 border-t" style={{ borderColor: "var(--line-soft)" }}>
   447→        <div>
   448→          <label className="label">Stage</label>
   449→          <select className="select" style={{ width: 170 }} value={stage} onChange={(e) => setStage(e.target.value)}>
   450→            {stages.map((s) => <option key={s.id}>{s.label}</option>)}
   451→          </select>
   452→        </div>
   453→        <div>
   454→          <label className="label">Bank override</label>
   455→          <select className="select" style={{ width: 140 }} value={bank} onChange={(e) => setBank(e.target.value)}>
   456→            <option>All</option>
   457→            {db.banks.map((b) => <option key={b.id}>{b.name}</option>)}
   458→          </select>
   459→        </div>
   460→        <div>
   461→          <label className="label">Max days</label>
   462→          <input className="input mono" style={{ width: 80 }} type="number" min={1} value={days} onChange={(e) => setDays(e.target.value)} />
   463→        </div>
   464→        <button className="btn btn-primary btn-sm" onClick={() => {
   465→          const d = Number(days);
   466→          if (!d || d < 1) return toast("error", "Enter a valid number of days.");
   467→          saveSla({ stage, bank: bank === "All" ? null : bank, maxDays: d, active: true });
   468→          toast("success", `SLA saved: ${stage} · ${bank === "All" ? "all banks" : bank} · ${d}d.`);
   469→        }}>
   470→          <IPlus size={13} /> Add rule
   471→        </button>
   472→      </div>
   473→    </div>
   474→  );
   475→}

   479→function DesignationsTab() {
   480→  const { db, addDesignation, updateDesignation, deleteDesignation, toast } = useStore();
   481→  const [name, setName] = useState("");
   482→  const [deleting, setDeleting] = useState<Designation | null>(null);
   483→
   484→  const FlagToggle = ({ on, label, onClick, locked }: { on: boolean; label: string; onClick: () => void; locked?: boolean }) => (
   485→    <button
   486→      type="button"
   487→      className="chip transition-all"
   488→      title={locked ? "Locked for this designation" : undefined}
   489→      style={
   490→        on
   491→          ? { background: "rgba(67,214,155,0.12)", borderColor: "rgba(67,214,155,0.5)", color: "var(--mint)" }
   492→          : { background: "var(--bg2)", borderColor: "var(--line)", color: "var(--ink-faint)" }
   493→      }
   494→      onClick={() => !locked && onClick()}
   495→    >
   496→      {label}
   497→    </button>
   498→  );
   499→
   500→  return (
   501→    <div className="card anim-fade-up">
   502→      <div className="px-4 py-3 border-b" style={{ borderColor: "var(--line-soft)" }}>
   503→        <h3 className="font-disp font-semibold text-[14px] m-0">Designations & permissions</h3>
   504→        <p className="text-[11.5px] text-[var(--ink-faint)] mt-0.5 mb-0">
   505→          Create your own designations beyond SPO / VRM. Scope decides what each holder sees: all cases, their team's, or only their own book.
   506→        </p>
   507→      </div>
   508→      <div className="divide-y" style={{ borderColor: "var(--line-soft)" }}>
   509→        {db.designations.map((d) => {
   510→          const holders = db.users.filter((u) => u.role === d.name).length;
   511→          return (
   512→            <div key={d.id} className="px-4 py-3 flex flex-wrap items-center gap-2.5" style={{ borderColor: "var(--line-soft)" }}>
   513→              {d.builtIn ? (
   514→                <span className="text-[13px] font-medium min-w-[150px]">{d.name}</span>
   515→              ) : (
   516→                <input
   517→                  className="input"
   518→                  style={{ width: 170, padding: "5px 9px" }}
   519→                  value={d.name}
   520→                  onChange={(e) => updateDesignation(d.id, { name: e.target.value })}
   521→                />
   522→              )}
   523→              {d.super && <Chip tone="amber">supreme</Chip>}
   524→              {d.builtIn && <Chip tone="slate">built-in</Chip>}
   525→              <select
   526→                className="select"
   527→                style={{ width: 130 }}
   528→                value={d.scope}
   529→                disabled={d.super}
   530→                onChange={(e) => updateDesignation(d.id, { scope: e.target.value as Designation["scope"] })}
   531→              >
   532→                <option value="all">sees all</option>
   533→                <option value="team">sees team</option>
   534→                <option value="own">own book</option>
   535→              </select>
   536→              <FlagToggle on={d.issueTasks} label="can issue tasks" locked={d.super} onClick={() => updateDesignation(d.id, { issueTasks: !d.issueTasks })} />
   537→              <FlagToggle on={d.admin} label="admin console" locked={d.super} onClick={() => updateDesignation(d.id, { admin: !d.admin })} />
   538→              <span className="text-[11.5px] text-[var(--ink-faint)]">{holders} holder{holders === 1 ? "" : "s"}</span>
   539→              {!d.builtIn && (
   540→                <button className="btn btn-danger btn-sm ml-auto" onClick={() => setDeleting(d)}>
   541→                  <ITrash size={13} />
   542→                </button>
   543→              )}
   544→            </div>
   545→          );
   546→        })}
   547→      </div>
   548→      <div className="flex flex-wrap items-end gap-2 px-4 py-3 border-t" style={{ borderColor: "var(--line-soft)" }}>
   549→        <div>
   550→          <label className="label">New designation</label>
   551→          <input className="input" style={{ width: 220 }} placeholder="e.g. Documents Officer" value={name} onChange={(e) => setName(e.target.value)} />
   552→        </div>
   553→        <button
   554→          className="btn btn-primary btn-sm"
   555→          onClick={() => {
   556→            const err = addDesignation(name);
   557→            if (err) return toast("error", err);
   558→            toast("success", `Designation "${name.trim()}" created — set its scope and permissions above.`);
   559→            setName("");
   560→          }}
   561→        >
   562→          <IPlus size={13} /> Add designation
   563→        </button>
   564→      </div>
   565→
   566→      <ConfirmModal
   567→        open={!!deleting}
   568→        onClose={() => setDeleting(null)}
   569→        title={`Delete "${deleting?.name}"?`}
   570→        body="Only designations with no holders can be deleted."
   571→        confirmLabel="Delete"
   572→        onConfirm={() => {
   573→          if (!deleting) return;
   574→          const err = deleteDesignation(deleting.id);
   575→          if (err) toast("error", err);
   576→          else toast("info", `Designation "${deleting.name}" deleted.`);
   577→        }}
   578→      />
   579→    </div>
   580→  );
   581→}
