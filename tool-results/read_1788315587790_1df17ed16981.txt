    13→/* a lightweight reply thread shared by directives */
    14→function DirectiveReply({ replies, onSend }: { replies: Reply[]; onSend: (t: string) => void }) {
    15→  const { userById } = useStore();
    16→  const [draft, setDraft] = useState("");
    17→  const send = () => {
    18→    if (!draft.trim()) return;
    19→    onSend(draft);
    20→    setDraft("");
    21→  };
    22→  return (
    23→    <div className="mt-2.5 space-y-2">
    24→      {replies.map((r) => (
    25→        <div key={r.id} className="flex items-start gap-2 anim-fade-in">
    26→          <Avatar name={userById(r.userId)?.name ?? "?"} size={20} />
    27→          <div className="min-w-0 flex-1 rounded-lg px-2.5 py-1.5" style={{ background: "var(--tint)", border: "1px solid var(--line-soft)" }}>
    28→            <span className="text-[11.5px] font-semibold">{userById(r.userId)?.name ?? "—"}</span>
    29→            <span className="mono text-[9.5px] text-[var(--ink-faint)] ml-1.5">{relTime(r.at)}</span>
    30→            <p className="text-[12px] text-[var(--ink-dim)] m-0 leading-snug">{r.text}</p>
    31→          </div>
    32→        </div>
    33→      ))}
    34→      <div className="flex items-center gap-2">
    35→        <input className="input" style={{ flex: 1, padding: "6px 10px", fontSize: 12.5 }} placeholder="Reply… (Enter)" value={draft}
    36→          onChange={(e) => setDraft(e.target.value)} onKeyDown={(e) => e.key === "Enter" && send()} />
    37→        <button className="btn btn-ghost btn-sm shrink-0" onClick={send} disabled={!draft.trim()}>Send</button>
    38→      </div>
    39→    </div>
    40→  );
    41→}

    44→function DirectivesBanner({ c }: { c: LoanCase }) {
    45→  const { db, session, userById, canInstruct, addInstruction, completeInstruction, replyToInstruction, completeBulletin, replyToBulletin, toast } = useStore();

    47→  const [openThread, setOpenThread] = useState<Record<string, boolean>>({});
    48→  const [showComposer, setShowComposer] = useState(false);
    49→  const [text, setText] = useState("");
    50→  const [assignee, setAssignee] = useState(c.ownerId);
    51→  const [due, setDue] = useState(inDaysISO(2));
    52→  const manager = canInstruct();
    53→  if (!me) return null;
    54→
    55→  type Dir =
    56→    | { kind: "instruction"; key: string; i: Instruction }
    57→    | { kind: "bulletin"; key: string; b: BulletinItem };
    58→
    59→  const instrs: Dir[] = db.instructions.filter((i) => i.caseId === c.id).map((i) => ({ kind: "instruction", key: `i${i.id}`, i }));
    60→  const bulls: Dir[] = db.bulletin
    61→    .filter((b) => b.caseId === c.id && !b.isTemplate && !b.dropped)
    62→    .map((b) => ({ kind: "bulletin", key: `b${b.id}`, b }));
    63→  const all = [...instrs, ...bulls].sort((a, z) => {
    64→    const ao = a.kind === "instruction" ? a.i.status : a.b.status;
    65→    const zo = z.kind === "instruction" ? z.i.status : z.b.status;
    66→    if (ao !== zo) return ao === "Open" ? -1 : 1;
    67→    const at = a.kind === "instruction" ? a.i.createdAt : a.b.createdAt;
    68→    const zt = z.kind === "instruction" ? z.i.createdAt : z.b.createdAt;
    69→    return zt.localeCompare(at);
    70→  });
    71→  const isActive = c.caseStatus === "Active";
    72→  if (all.length === 0 && !manager) return null;
    73→
    74→  const openCount = all.filter((d) => (d.kind === "instruction" ? d.i.status : d.b.status) === "Open").length;
    75→  const f = flagsFor(db.designations, me.role);
    76→
    77→  const canActInstr = (i: Instruction) => f.scope === "all" || i.issuedBy === me.id || i.assignedTo === me.id;
    78→  const canActBull = (b: BulletinItem) => bulletinCanAct(b, me, db);

    82→      <div className="flex items-center gap-2.5 px-4 py-2.5" style={{ borderBottom: "1px solid color-mix(in srgb, var(--amber) 20%, transparent)" }}>
    83→        <IFlag size={16} className="text-[var(--amber)]" />
    84→        <span className="font-disp font-semibold text-[13.5px]" style={{ color: "var(--amber)" }}>Directives on this case</span>
    85→        <span className="mono text-[11px] px-1.5 py-0.5 rounded-full" style={{ background: "var(--amber-tint)", color: "var(--amber)" }}>
    86→          {openCount} open
    87→        </span>
    88→      </div>
    89→      <div className="divide-y" style={{ borderColor: "color-mix(in srgb, var(--amber) 14%, transparent)" }}>
    90→        {all.map((d) => {
    91→          const isInstr = d.kind === "instruction";
    92→          const status = isInstr ? d.i.status : d.b.status;
    93→          const createdAt = isInstr ? d.i.createdAt : d.b.createdAt;
    94→          const issuerId = isInstr ? d.i.issuedBy : d.b.issuedBy;
    95→          const text = isInstr ? d.i.instruction : d.b.task;
    96→          const replies = isInstr ? d.i.replies : d.b.replies;
    97→          const canAct = status === "Open" && (isInstr ? canActInstr(d.i) : canActBull(d.b));
    98→          const issuer = userById(issuerId);
    99→          const showThread = !!openThread[d.key];
   100→          const done = status === "Done";
   101→          return (
   102→            <div key={d.key} className="px-4 py-3" style={{ opacity: done ? 0.65 : 1 }}>
   103→              <div className="flex items-start gap-3">
   104→                <Avatar name={issuer?.name ?? "?"} size={30} />
   105→                <div className="min-w-0 flex-1">
   106→                  <div className="flex flex-wrap items-baseline gap-x-2">
   107→                    <span className="font-disp font-semibold text-[13px]">{issuer?.name ?? "—"}</span>
   108→                    <Chip tone={isInstr ? "sky" : "amber"}>{isInstr ? "instruction" : "bulletin"}</Chip>
   109→                    {!isInstr && d.b.carriedFrom && <Chip tone="sky">carried from {fmtDate(d.b.carriedFrom)}</Chip>}
   110→                    {!isInstr && d.b.templateId && <Chip tone="slate">routine</Chip>}
   111→                    <span className="mono text-[10.5px] text-[var(--ink-faint)]">{relTime(createdAt)}</span>
   112→                    {isInstr && <DueChip dueISO={d.i.dueDate} />}
   113→                  </div>
   114→                  <p className={`text-[13.5px] leading-relaxed mt-1 mb-0 ${done ? "line-through text-[var(--ink-faint)]" : ""}`}>{text}</p>
   115→                  {isInstr && (
   116→                    <p className="text-[11px] text-[var(--ink-faint)] mt-0.5 mb-0">assigned to <span className="text-[var(--ink-dim)]">{userById(d.i.assignedTo)?.name ?? "—"}</span></p>
   117→                  )}
   118→                  {done && (isInstr ? d.i.completedAt : d.b.completedAt) && (
   119→                    <p className="text-[11px] mt-0.5 mb-0" style={{ color: "var(--mint)" }}>✓ done {relTime((isInstr ? d.i.completedAt : d.b.completedAt) as string)}</p>
   120→                  )}
   121→                </div>
   122→                <div className="flex flex-col gap-1.5 shrink-0">
   123→                  {canAct && (isInstr || !d.b.caseId) && (
   124→                    <button className="btn btn-mint btn-sm" onClick={() => {
   125→                      if (isInstr) { completeInstruction(d.i.id); } else { completeBulletin(d.b.id); }

   127→                    }}>
   128→                      <ICheck size={13} /> Done
   129→                    </button>

   131→                  {canAct && !isInstr && d.b.caseId === c.id && (

   155→                </div>
   156→              )}
   157→            </div>
   158→          );

   160→        {all.length === 0 && (
   161→          <p className="px-4 py-3 text-[12.5px] text-[var(--ink-faint)] m-0">No directives yet — issue the first one below.</p>
   162→        )}
   163→      </div>

   192→        </div>
   193→      )}
   194→    </div>
   195→  );
   196→}

   440→            {session?.role === "Head of Company" && (

   476→                      {s.label}
   477→                    </span>
   478→                  </button>
   479→                  {i < stages.length - 1 && (
   480→                    <div className="w-7 h-px mx-1 mb-[22px]" style={{ background: i < stageIdx ? "color-mix(in srgb, var(--mint) 52%, transparent)" : "var(--line)" }} />

   491→          This case is <strong style={{ color: c.caseStatus === "Lost" ? "var(--coral)" : "var(--sky)" }}>{c.caseStatus === "Lost" ? "lost" : "booked"}</strong> — the task engine is paused. Reopen it to resume work.
   492→        </div>
   493→      )}
   494→
   495→      {/* task + side panels */}
   496→      <div className="grid grid-cols-1 xl:grid-cols-[1fr_360px] gap-4 items-start">

   503→              <button className="btn btn-primary mx-auto" onClick={() => setShowNext(true)}>
   504→                <IPlus size={15} /> Open next task
   505→              </button>
   506→            </div>
   507→          )}
   508→          {isActive && (
   509→            <div className="flex justify-end">
   510→              {openTask && (
   511→                <button className="btn btn-ghost" onClick={() => setShowNext(true)}>
   512→                  <IArrowR size={15} /> Open next task
   513→                </button>
   514→              )}
   515→            </div>
   516→          )}
   517→

   519→          <div className="card anim-fade-up">
   520→            <div className="px-4 py-3 border-b" style={{ borderColor: "var(--line-soft)" }}>
   521→              <h3 className="font-disp font-semibold text-[13.5px] m-0">Task history</h3>

   537→                      </div>
   538→                    </div>
   539→                    <Chip tone="slate">{t.waitingFor}</Chip>
   540→                  </div>
   541→                ))}
   542→              </div>

   701→      />
   702→    </div>
   703→  );
   704→}
