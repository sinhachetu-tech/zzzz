    67→/* migrate saved payloads from older shapes so reloading a historic check never crashes */
    68→function normalizeInput(raw: Partial<MortgageInput>): MortgageInput {
    69→  const base = defaultInput();
    70→  const co = raw.coBorrower as
    71→    | (Partial<CoBorrower> & { existingEmis?: number })
    72→    | null
    73→    | undefined;
    74→  const coBorrower: CoBorrower | null = co
    75→    ? {
    76→        name: co.name ?? "",
    77→        dob: co.dob ?? "1992-01-15",
    78→        incomes: Array.isArray(co.incomes) ? co.incomes : [newIncomeRow()],
    79→        liabilities: Array.isArray(co.liabilities)
    80→          ? co.liabilities
    81→          : co.existingEmis
    82→          ? [{ ...newLiabRow("Other Loan"), name: "Carried-over EMIs", method: "Manual", monthlyEmi: co.existingEmis }]
    83→          : [],
    84→      }
    85→    : null;
    86→  return {
    87→    ...base,
    88→    ...raw,
    89→    customLtv: typeof raw.customLtv === "string" ? raw.customLtv : "",
    90→    coBorrower,
    91→    incomes: Array.isArray(raw.incomes) && raw.incomes.length ? raw.incomes : base.incomes,
    92→    liabilities: Array.isArray(raw.liabilities) ? raw.liabilities : [],
    93→  } as MortgageInput;
    94→}
