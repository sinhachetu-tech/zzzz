    11→function ReplyThread({ replies, onSend }: { replies: Reply[]; onSend: (text: string) => void }) {
    12→  const { userById } = useStore();
    13→  const [draft, setDraft] = useState("");
    14→  const send = () => {
    15→    if (!draft.trim()) return;
    16→    onSend(draft);
    17→    setDraft("");
    18→  };
    19→  return (
    20→    <div className="mt-3 space-y-2">
    21→      {replies.map((r) => (
    22→        <div key={r.id} className="flex items-start gap-2 anim-fade-in">
    23→          <Avatar name={userById(r.userId)?.name ?? "?"} size={22} />
    24→          <div className="min-w-0 flex-1 rounded-lg px-3 py-2" style={{ background: "var(--tint)", border: "1px solid var(--line-soft)" }}>
    25→            <div className="flex items-baseline gap-2">
    26→              <span className="text-[12px] font-semibold">{userById(r.userId)?.name ?? "—"}</span>
    27→              <span className="mono text-[10px] text-[var(--ink-faint)]">{relTime(r.at)}</span>
    28→            </div>
    29→            <p className="text-[12.5px] text-[var(--ink-dim)] m-0 mt-0.5 leading-snug">{r.text}</p>
    30→          </div>
    31→        </div>
    32→      ))}
    33→      <div className="flex items-center gap-2">
    34→        <input
    35→          className="input"
    36→          style={{ flex: 1 }}
    37→          placeholder="Reply… (press Enter)"
    38→          value={draft}
    39→          onChange={(e) => setDraft(e.target.value)}
    40→          onKeyDown={(e) => e.key === "Enter" && send()}
    41→        />
    42→        <button className="btn btn-ghost btn-sm shrink-0" onClick={send} disabled={!draft.trim()}>
    43→          Send
    44→        </button>
    45→      </div>
    46→    </div>
    47→  );
    48→}

    50→function StatusPill({ b }: { b: BulletinItem }) {
    51→  const today = todayISO();
    52→  if (b.dropped) return <Chip tone="slate">dropped</Chip>;
    53→  if (b.status === "Done") return <Chip tone="mint">done</Chip>;
    54→  if (b.date < today) return <Chip tone="coral">missed</Chip>;
    55→  return <Chip tone="amber">open</Chip>;
    56→}

   184→
   185→function Composer({ onIssued }: { onIssued: () => void }) {
   186→  const { db, session, issueBulletin, toast, visibleCases } = useStore();
   187→  const me = session!;
   188→  const f = flagsFor(db.designations, me.role);
   189→  const teamUsers = db.users.filter((u) => u.active && u.id !== me.id && (f.scope === "all" || u.team === me.team));
   190→  const [text, setText] = useState("");
   191→  const [targets, setTargets] = useState<number[]>(teamUsers.map((u) => u.id));
   192→  const [caseId, setCaseId] = useState<number | null>(null);
   193→  const [repeat, setRepeat] = useState<"none" | "daily" | "weekdays">("none");
   194→  const cases = visibleCases().filter((c) => c.caseStatus === "Active");
   195→
   196→  const toggle = (id: number) => setTargets((t) => (t.includes(id) ? t.filter((x) => x !== id) : [...t, id]));
   197→
   198→  const issue = () => {
   199→    if (!text.trim()) return toast("error", "Write the directive first.");
   200→    if (targets.length === 0) return toast("error", "Pick at least one teammate.");
   201→    issueBulletin({ date: todayISO(), task: text, caseId, targets, repeat, asTemplate: repeat !== "none" });
   202→    toast("success", repeat === "none" ? "Directive issued for today." : `Routine saved — issued today and every ${repeat === "daily" ? "day" : "working day (Mon–Fri)"}.`);
   203→    setText("");
   204→    setCaseId(null);
   205→    setRepeat("none");
   206→    onIssued();
   207→  };
   208→
   209→  return (
   210→    <div className="card p-4 anim-fade-up" style={{ borderColor: "var(--amber-line)" }}>
   211→      <div className="flex items-center gap-2 mb-3">
   212→        <IFlag size={15} className="text-[var(--amber)]" />
   213→        <h2 className="font-disp font-semibold text-[14px] m-0">Issue a directive</h2>
   214→        <span className="text-[11.5px] text-[var(--ink-faint)] ml-auto hidden sm:inline">lands on your team's morning bulletin</span>
   215→      </div>
   216→      <textarea
   217→        className="textarea"
   218→        rows={2}
   219→        autoFocus
   220→        placeholder="e.g. Every file stuck in Pre-Approval more than 5 days gets an RM call before noon — report back here."
   221→        value={text}
   222→        onChange={(e) => setText(e.target.value)}
   223→      />
   224→      <div className="flex flex-wrap items-center gap-2 mt-3">
   225→        <select className="select" style={{ width: 210 }} value={caseId ?? ""} onChange={(e) => setCaseId(e.target.value ? parseInt(e.target.value, 10) : null)}>
   226→          <option value="">General — no case link</option>
   227→          {cases.map((c) => (
   228→            <option key={c.id} value={c.id}>
   229→              {c.caseNumber} · {c.customer}
   230→            </option>
   231→          ))}
   232→        </select>
   233→        <select className="select" style={{ width: 190 }} value={repeat} onChange={(e) => setRepeat(e.target.value as typeof repeat)}>
   234→          <option value="none">One-off (today only)</option>
   235→          <option value="daily">Repeat · every day</option>
   236→          <option value="weekdays">Repeat · Mon–Fri</option>
   237→        </select>
   238→        <div className="flex items-center gap-1.5 ml-auto">
   239→          <span className="text-[11px] uppercase tracking-[0.08em] text-[var(--ink-faint)] font-disp font-semibold">to</span>
   240→          {teamUsers.map((u) => (
   241→            <button
   242→              key={u.id}
   243→              className="chip transition-all"
   244→              style={
   245→                targets.includes(u.id)
   246→                  ? { background: "var(--amber-tint)", borderColor: "var(--amber)", color: "var(--amber)" }
   247→                  : { background: "var(--bg2)", borderColor: "var(--line)", color: "var(--ink-faint)" }
   248→              }
   249→              onClick={() => toggle(u.id)}
   250→            >
   251→              {u.name.split(" ")[0]}
   252→            </button>
   253→          ))}
   254→        </div>
   255→      </div>
   256→      <div className="flex items-center justify-between mt-3">
   257→        <p className="text-[11px] text-[var(--ink-faint)] m-0">
   258→          Case-linked directives auto-resolve when the case is booked or lost. Routines spawn a fresh item each morning.
   259→        </p>
   260→        <button className="btn btn-primary btn-sm shrink-0" onClick={issue}>
   261→          <IPlus size={13} /> Issue{repeat !== "none" ? " routine" : ""}
   262→        </button>
   263→      </div>
   264→    </div>
   265→  );
   266→}

   270→function Scoreboard({ items }: { items: BulletinItem[] }) {
   271→  const { db, userById } = useStore();
   272→  const rows = useMemo(() => {
   273→    const map = new Map<number, { done: number; total: number }>();
   274→    for (const b of items) {
   275→      if (b.isTemplate || b.dropped) continue;
   276→      for (const t of b.targets) {
   277→        const cur = map.get(t) ?? { done: 0, total: 0 };
   278→        cur.total += 1;
   279→        if (b.status === "Done") cur.done += 1;
   280→        map.set(t, cur);
   281→      }
   282→    }
   283→    return [...map.entries()]
   284→      .map(([id, v]) => ({ id, name: userById(id)?.name ?? "?", ...v }))
   285→      .sort((a, b) => b.total - a.total);
   286→  }, [items, userById]);
   287→
   288→  if (rows.length === 0) return null;
   289→  return (
   290→    <div className="card p-4 anim-fade-up">
   291→      <h3 className="font-disp font-semibold text-[13.5px] m-0 mb-1">Team pulse · today</h3>
   292→      <p className="text-[11px] text-[var(--ink-faint)] m-0 mb-3">directives cleared per teammate</p>
   293→      <div className="space-y-2.5">
   294→        {rows.map((r) => {
   295→          const pct = r.total ? (r.done / r.total) * 100 : 0;
   296→          const clean = r.done === r.total && r.total > 0;
   297→          return (
   298→            <div key={r.id} className="flex items-center gap-2.5">
   299→              <Avatar name={r.name} size={24} />
   300→              <div className="flex-1 min-w-0">
   301→                <div className="flex justify-between items-baseline mb-0.5">
   302→                  <span className="text-[12px] truncate">{r.name}</span>
   303→                  <span className="mono text-[11px]" style={{ color: clean ? "var(--mint)" : pct < 50 ? "var(--coral)" : "var(--ink-dim)" }}>
   304→                    {r.done}/{r.total}
   305→                  </span>
   306→                </div>
   307→                <div className="h-[5px] rounded-full overflow-hidden" style={{ background: "var(--track)" }}>
   308→                  <div className="h-full rounded-full" style={{ width: `${pct}%`, background: clean ? "var(--mint)" : pct < 50 ? "var(--coral)" : "var(--amber)", transition: "width 0.7s cubic-bezier(0.22,1,0.36,1)" }} />
   309→                </div>
   310→              </div>
   311→            </div>
   312→          );
   313→        })}
   314→      </div>
   315→    </div>
   316→  );
   317→}

   321→type Tab = "today" | "missed" | "archive";

   323→export default function Bulletin() {

   325→  const me = session;
   326→  const [tab, setTab] = useState<Tab>("today");
   327→  const [deleting, setDeleting] = useState<BulletinItem | null>(null);
   328→  const [showComposer, setShowComposer] = useState(true);
   329→
   330→  const today = todayISO();
   331→  const manager = canInstruct();
   332→
   333→  const visible = useMemo(
   334→    () => (me ? db.bulletin.filter((b) => !b.isTemplate && bulletinVisible(b, me, db)) : []),
   335→    [db, me]
   336→  );
   337→  const todayItems = visible
   338→    .filter((b) => b.date === today)
   339→    .sort((a, b) => (a.status === b.status ? a.createdAt.localeCompare(b.createdAt) : a.status === "Open" ? -1 : 1));

   341→    .filter((b) => b.date < today && b.status === "Open" && !b.dropped)
   342→    .sort((a, b) => b.date.localeCompare(a.date));
   343→  const archiveItems = visible
   344→    .filter((b) => b.date < today || b.dropped || (b.date === today && b.status === "Done"))
   345→    .sort((a, b) => b.date.localeCompare(a.date) || b.createdAt.localeCompare(a.createdAt));

   347→    () => (me ? db.bulletin.filter((b) => b.isTemplate && (bulletinCanManage(b, me, db) || bulletinVisible(b, me, db))) : []),
   348→    [db, me]
   349→  );
   350→
   351→  if (!me) return null;
   352→
   353→  const archiveGroups = new Map<string, BulletinItem[]>();

   355→    const list = archiveGroups.get(b.date) ?? [];
   356→    list.push(b);
   357→    archiveGroups.set(b.date, list);

   370→        b.targets.map((t) => userById(t)?.name ?? "").join("; "),
   371→        b.caseId ? db.cases.find((c) => c.id === b.caseId)?.caseNumber ?? "" : "",
   372→        b.dropped ? "Dropped" : b.status,
   373→        b.completedAt ? relTime(b.completedAt) : "",

   375→    );
   376→    toast("success", "Archive exported to CSV.");

   379→  const tabs: { key: Tab; label: string; count: number; alert?: boolean }[] = [
   380→    { key: "today", label: "Today", count: todayItems.filter((b) => b.status === "Open" && !b.dropped).length },
   381→    { key: "missed", label: "Missed", count: missedItems.length, alert: missedItems.length > 0 },
   382→    { key: "archive", label: "Archive", count: archiveItems.length },

   387→      <div className="flex flex-wrap items-end justify-between gap-3">

   389→          <h1 className="font-disp font-bold text-[24px] tracking-tight m-0 flex items-center gap-2.5">
   390→            <IFlag size={22} className="text-[var(--amber)]" /> Morning Bulletin
   391→          </h1>
   392→          <p className="text-[13px] text-[var(--ink-dim)] mt-0.5 mb-0">
   393→            Directives for {new Date().toLocaleDateString("en-GB", { weekday: "long", day: "numeric", month: "long" })} · issue once, the team answers with Done or a reply
   394→          </p>
   395→        </div>
   396→        {manager && (
   397→          <button className="btn btn-ghost btn-sm" onClick={() => setShowComposer((s) => !s)}>

   404→
   405→      <div className="flex items-center gap-2">

   409→            className="chip transition-all"

   413→                  ? { background: "color-mix(in srgb, var(--coral) 14%, transparent)", borderColor: "var(--coral)", color: "var(--coral)" }
   414→                  : { background: "var(--amber-tint)", borderColor: "var(--amber)", color: "var(--amber)" }
   415→                : { background: "var(--bg2)", borderColor: "var(--line)", color: "var(--ink-faint)" }
   416→            }
   417→            onClick={() => setTab(t.key)}
   418→          >

   420→            <span className="mono font-semibold">{t.count}</span>
   421→          </button>

   448→
   449→      {tab === "missed" && (
   450→        <div className="space-y-3">
   451→          {missedItems.length === 0 ? (
   452→            <div className="card p-8">
   453→              <EmptyState
   454→                icon={<ICheck size={26} />}
   455→                title="Nothing slipped through"
   456→                body="Every past directive was either completed, carried forward, or dropped. That's the loop working."
   457→              />
   458→            </div>
   459→          ) : (
   460→            <>
   461→              <div className="card px-4 py-3 text-[13px] anim-fade-up" style={{ borderColor: "color-mix(in srgb, var(--coral) 40%, transparent)" }}>
   462→                <strong style={{ color: "var(--coral)" }}>{missedItems.length} directive{missedItems.length === 1 ? "" : "s"}</strong>{" "}
   463→                <span className="text-[var(--ink-dim)]">
   464→                  went past their day unfinished. Carry them forward to today's bulletin — or drop them and keep the record clean.
   465→                </span>
   466→              </div>
   467→              {missedItems.map((b) => (
   468→                <DirectiveCard key={b.id} b={b} />
   469→              ))}
   470→            </>
   471→          )}
   472→        </div>
   473→      )}
   474→
   475→      {tab === "archive" && (
   476→        <div className="space-y-4">
   477→          {templates.length > 0 && (
   478→            <div className="card anim-fade-up">
   479→              <div className="px-4 py-3 border-b flex items-center gap-2" style={{ borderColor: "var(--line-soft)" }}>
   480→                <ICalendar size={14} className="text-[var(--amber)]" />
   481→                <h3 className="font-disp font-semibold text-[13.5px] m-0">Recurring routines</h3>
   482→                <span className="text-[11px] text-[var(--ink-faint)] ml-auto">spawn a fresh item every working morning</span>
   483→              </div>
   484→              <div className="divide-y" style={{ borderColor: "var(--line-soft)" }}>
   485→                {templates.map((t) => (
   486→                  <div key={t.id} className="px-4 py-3 flex flex-wrap items-center gap-2.5" style={{ borderColor: "var(--line-soft)" }}>
   487→                    <Chip tone="amber">{t.repeat === "daily" ? "daily" : "Mon–Fri"}</Chip>
   488→                    <span className="text-[13px] flex-1 min-w-[220px] truncate">{t.task}</span>
   489→                    <span className="text-[11px] text-[var(--ink-faint)]">
   490→                      by {userById(t.issuedBy)?.name ?? "—"} · since {fmtDate(t.date)} · to {t.targets.map((x) => userById(x)?.name.split(" ")[0] ?? "?").join(", ")}
   491→                    </span>
   492→                    {bulletinCanDelete(t, me, db) && (
   493→                      <button className="btn btn-danger btn-sm" onClick={() => setDeleting(t)}>
   494→                        <ITrash size={13} /> Retire
   495→                      </button>
   496→                    )}
   497→                  </div>
   498→                ))}
   499→              </div>
   500→            </div>
   501→          )}
   502→
   503→          {groupDates.length === 0 ? (
   504→            <div className="card p-8">
   505→              <EmptyState icon={<IHistory size={26} />} title="Archive is empty" body="Completed, carried, dropped and auto-resolved directives collect here, date by date." />
   506→            </div>
   507→          ) : (
   508→            groupDates.map((date) => {
   509→              const items = archiveGroups.get(date)!;
   510→              const daysAgo = daysBetween(date, today);
   511→              return (
   512→                <div key={date}>
   513→                  <div className="flex items-baseline gap-2.5 px-1 mb-2">
   514→                    <span className="font-disp font-semibold text-[13px]">{daysAgo === 1 ? "Yesterday" : fmtDate(date)}</span>
   515→                    <span className="mono text-[10.5px] text-[var(--ink-faint)]">{daysAgo}d ago</span>
   516→                    <span className="text-[11px] text-[var(--ink-faint)]">
   517→                      {items.filter((i) => i.status === "Done" && !i.dropped).length} done · {items.filter((i) => i.dropped).length} dropped
   518→                    </span>
   519→                    <span className="flex-1 h-px" style={{ background: "var(--line-soft)" }} />
   520→                  </div>
   521→                  <div className="space-y-2.5">
   522→                    {items.map((b) => (
   523→                      <ArchiveRow key={b.id} b={b} onDelete={() => bulletinCanDelete(b, me, db) && setDeleting(b)} canDelete={bulletinCanDelete(b, me, db)} />
   524→                    ))}
   525→                  </div>
   526→                </div>
   527→              );
   528→            })
   529→          )}
   530→        </div>
   531→      )}
   532→
   533→      <ConfirmModal
   534→        open={!!deleting}
   535→        onClose={() => setDeleting(null)}
   536→        title={deleting?.isTemplate ? "Retire this routine?" : "Delete this archive entry?"}
   537→        body={
   538→          deleting?.isTemplate ? (
   539→            <span>The routine stops spawning future items. <strong>Past instances stay in the archive</strong> as the record.</span>
   540→          ) : (
   541→            <span>Removes the directive and its replies from the archive permanently.</span>
   542→          )
   543→        }
   544→        confirmLabel={deleting?.isTemplate ? "Retire routine" : "Delete"}
   545→        onConfirm={() => {
   546→          if (!deleting) return;
   547→          deleteBulletin(deleting.id);
   548→          toast("info", deleting.isTemplate ? "Routine retired — no more spawns." : "Archive entry deleted.");
   549→        }}
   550→      />
   551→    </div>
   552→  );
   553→}

   555→function ArchiveRow({ b, onDelete, canDelete }: { b: BulletinItem; onDelete: () => void; canDelete: boolean }) {
   556→  const { db, session, nav, userById } = useStore();
   557→  const me = session;
   558→  const [showThread, setShowThread] = useState(false);
   559→  if (!me) return null;
   560→  const issuer = userById(b.issuedBy);
   561→  const linkedCase = b.caseId ? db.cases.find((c) => c.id === b.caseId) : null;
   562→  return (
   563→    <div className="card p-3.5 anim-fade-up" style={{ opacity: 0.85, borderLeft: `3px solid ${b.dropped ? "var(--line)" : "var(--mint)"}` }}>
   564→      <div className="flex items-start gap-3">
   565→        <Avatar name={issuer?.name ?? "?"} size={26} />
   566→        <div className="min-w-0 flex-1">
   567→          <div className="flex flex-wrap items-baseline gap-x-2">
   568→            <span className="font-disp font-semibold text-[12.5px]">{issuer?.name ?? "—"}</span>
   569→            <span className="mono text-[10.5px] text-[var(--ink-faint)]">{relTime(b.createdAt)}</span>
   570→            <StatusPill b={b} />
   571→            {b.carriedFrom && <Chip tone="sky">carried from {fmtDate(b.carriedFrom)}</Chip>}
   572→            {linkedCase && (
   573→              <button className="chip transition-colors hover:opacity-80" style={{ background: "var(--tint)", borderColor: "var(--line)", color: "var(--ink-dim)", cursor: "pointer" }} onClick={() => nav({ name: "case", id: linkedCase.id })}>
   574→                <span className="mono" style={{ color: "var(--amber)" }}>{linkedCase.caseNumber}</span>
   575→              </button>
   576→            )}
   577→          </div>
   578→          <p className="text-[13px] text-[var(--ink-dim)] m-0 mt-1 leading-snug">{b.task}</p>
   579→          <p className="text-[11px] text-[var(--ink-faint)] m-0 mt-0.5">
   580→            to {b.targets.map((t) => userById(t)?.name.split(" ")[0] ?? "?").join(", ")}
   581→            {b.status === "Done" && b.completedAt && ` · done ${relTime(b.completedAt)}${b.completedBy ? ` by ${userById(b.completedBy)?.name ?? ""}` : " (auto)"}`}
   582→          </p>
   583→        </div>
   584→        <div className="flex items-center gap-1.5 shrink-0">
   585→          {b.replies.length > 0 && (
   586→            <button className="btn btn-ghost btn-sm" onClick={() => setShowThread((s) => !s)}>
   587→              Replies ({b.replies.length})
   588→            </button>
   589→          )}
   590→          {canDelete && (
   591→            <button className="btn btn-ghost btn-sm" onClick={onDelete} title="Delete from archive">
   592→              <ITrash size={13} />
   593→            </button>
   594→          )}
   595→        </div>
   596→      </div>
   597→      {showThread && (
   598→        <div className="pl-[38px] mt-2 space-y-2">
   599→          {b.replies.map((rp) => (
   600→            <div key={rp.id} className="rounded-lg px-3 py-2 anim-fade-in" style={{ background: "var(--tint)", border: "1px solid var(--line-soft)" }}>
   601→              <div className="flex items-baseline gap-2">
   602→                <span className="text-[12px] font-semibold">{userById(rp.userId)?.name ?? "—"}</span>
   603→                <span className="mono text-[10px] text-[var(--ink-faint)]">{relTime(rp.at)}</span>
   604→              </div>
   605→              <p className="text-[12.5px] text-[var(--ink-dim)] m-0 mt-0.5">{rp.text}</p>
   606→            </div>
   607→          ))}
   608→        </div>
   609→      )}
   610→    </div>
   611→  );
   612→}
